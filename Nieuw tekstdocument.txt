    <script>
        let virgilProtectionActive = true;

        // Dynamic Random IP Generator
        function generateRandomIP() {
            return Math.floor(Math.random() * 225 + 10) + "." +
                   Math.floor(Math.random() * 255) + "." +
                   Math.floor(Math.random() * 255) + "." +
                   Math.floor(Math.random() * 254 + 1);
        }

        function rotateIP() {
            const ipDisplay = document.getElementById('ip-address');
            ipDisplay.style.opacity = '0.3';
            setTimeout(() => {
                ipDisplay.innerText = generateRandomIP();
                ipDisplay.style.opacity = '1';
            }, 150);
        }

        // Automatic IP Shift
        setInterval(() => {
            rotateIP();
        }, 10000);

        // Virgil Barrier Toggle Switch
        function toggleVirgilProtection(checkbox) {
            virgilProtectionActive = checkbox.checked;
            const statusText = document.getElementById('virgil-status-text');
            const shieldStatus = document.getElementById('shield-status');

            if (virgilProtectionActive) {
                statusText.innerText = "AAN";
                statusText.style.color = "var(--weed-accent)";
                shieldStatus.innerText = "Actief";
            } else {
                statusText.innerText = "UIT";
                statusText.style.color = "var(--red-alert)";
                shieldStatus.innerText = "Gevaar";
            }
        }

        // Intercept Simualtion
        function simuleerVirgilAanval() {
            if (virgilProtectionActive) {
                document.getElementById('virgil-alert').style.display = 'flex';
            } else {
                alert("Waarschuwing: Virgil Barrier staat uit! Virgil Burger heeft je getrackt.");
            }
        }

        function clearVirgilAlert() {
            document.getElementById('virgil-alert').style.display = 'none';
            rotateIP();
        }
    </script>
